cmake_minimum_required(VERSION 3.16)
project(chimaera_m5_benchmark LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(latency_endpoint endpoint.cpp)
ament_target_dependencies(latency_endpoint rclcpp sensor_msgs rosgraph_msgs std_msgs)
target_link_libraries(latency_endpoint dl)
install(TARGETS latency_endpoint DESTINATION lib/chimaera_ros_bridge)
