cmake_minimum_required(VERSION 3.16)
project(ball_catching_robot)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(orocos_kdl REQUIRED)
find_package(urdf REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSONC REQUIRED IMPORTED_TARGET json-c)

add_library(catching_core STATIC src/model.cpp src/motion.cpp src/vision.cpp)
target_compile_features(catching_core PUBLIC cxx_std_17)
target_compile_options(catching_core PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(catching_core PUBLIC include ${orocos_kdl_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
target_link_libraries(catching_core PUBLIC Eigen3::Eigen ${orocos_kdl_LIBRARIES} ${OpenCV_LIBRARIES})
ament_target_dependencies(catching_core PUBLIC urdf)
foreach(node perception intercept effort_control)
  add_executable(${node} src/${node}.cpp)
  target_compile_options(${node} PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(${node} catching_core PkgConfig::JSONC)
  ament_target_dependencies(${node} rclcpp sensor_msgs nav_msgs geometry_msgs trajectory_msgs std_msgs)
  install(TARGETS ${node} DESTINATION lib/${PROJECT_NAME})
endforeach()
add_executable(validate_model src/validate_model.cpp)
target_link_libraries(validate_model catching_core)
install(TARGETS validate_model DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/create_fixture.py
                  ${CMAKE_CURRENT_BINARY_DIR}/test_robot.urdf RESULT_VARIABLE fixture_status)
  if(NOT fixture_status EQUAL 0)
    message(FATAL_ERROR "Could not generate FR3 test fixture")
  endif()
  ament_add_gtest(core_test test/core.cpp)
  target_link_libraries(core_test catching_core PkgConfig::JSONC)
  target_compile_definitions(core_test PRIVATE TEST_URDF="${CMAKE_CURRENT_BINARY_DIR}/test_robot.urdf"
    PYTHON_REFERENCE="${CMAKE_CURRENT_SOURCE_DIR}/test/python_reference.json")
endif()
ament_package()
