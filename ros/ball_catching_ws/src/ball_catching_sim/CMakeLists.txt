cmake_minimum_required(VERSION 3.16)
project(ball_catching_sim)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSON_C REQUIRED IMPORTED_TARGET json-c)
find_package(ignition-gazebo6 REQUIRED)
find_package(ignition-plugin1 REQUIRED COMPONENTS register)
add_library(ball_catching_host SHARED src/host.cpp)
target_compile_features(ball_catching_host PUBLIC cxx_std_17)
target_compile_options(ball_catching_host PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(ball_catching_host PRIVATE include)
target_link_libraries(ball_catching_host ignition-gazebo6::ignition-gazebo6 ignition-plugin1::register PkgConfig::JSON_C)
ament_target_dependencies(ball_catching_host rclcpp sensor_msgs std_msgs std_srvs geometry_msgs)
ament_python_install_package(${PROJECT_NAME})
install(TARGETS ball_catching_host LIBRARY DESTINATION lib)
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  enable_testing()
  add_executable(command_replay_test test/command_replay.cpp)
  target_include_directories(command_replay_test PRIVATE include)
  target_compile_features(command_replay_test PRIVATE cxx_std_17)
  target_compile_options(command_replay_test PRIVATE -UNDEBUG)
  add_test(NAME command_replay COMMAND command_replay_test)
  add_executable(scoring_test test/scoring.cpp)
  target_include_directories(scoring_test PRIVATE include)
  target_compile_options(scoring_test PRIVATE -UNDEBUG)
  add_test(NAME scoring COMMAND scoring_test)
  add_test(NAME launch_stepping COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/test/launch_stepping.py
    --plugin-dir $<TARGET_FILE_DIR:ball_catching_host>
    --output ${CMAKE_CURRENT_BINARY_DIR}/launch-stepping-validation --require-validation)
  set_tests_properties(launch_stepping PROPERTIES TIMEOUT 120)
  add_test(NAME checkpoint_replay COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/test/checkpoint_replay.py
    --plugin-dir $<TARGET_FILE_DIR:ball_catching_host>
    --output ${CMAKE_CURRENT_BINARY_DIR}/checkpoint-replay-validation)
  set_tests_properties(checkpoint_replay PROPERTIES TIMEOUT 90 RUN_SERIAL TRUE)
endif()
ament_package()
