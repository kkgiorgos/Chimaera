cmake_minimum_required(VERSION 3.16)
project(wall_follow_benchmark)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  src/controller.cpp include/wall_follow_benchmark/core.hpp)
file(SHA256 ${CMAKE_CURRENT_SOURCE_DIR}/src/controller.cpp NODE_HASH)
file(SHA256 ${CMAKE_CURRENT_SOURCE_DIR}/include/wall_follow_benchmark/core.hpp CORE_HASH)
add_executable(controller src/controller.cpp)
target_compile_features(controller PRIVATE cxx_std_17)
target_compile_options(controller PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(controller PRIVATE NODE_HASH="${NODE_HASH}" CORE_HASH="${CORE_HASH}")
target_include_directories(controller PRIVATE include)
ament_target_dependencies(controller rclcpp rcl_interfaces geometry_msgs sensor_msgs nav_msgs)
install(TARGETS controller DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
ament_package()
