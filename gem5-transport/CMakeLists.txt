cmake_minimum_required(VERSION 3.21)
project(Gem5Transport LANGUAGES CXX)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The gem5 transport requires Linux")
endif()

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_library(chimaera_transport INTERFACE)
add_library(chimaera::transport ALIAS chimaera_transport)
target_compile_features(chimaera_transport INTERFACE cxx_std_20)
target_include_directories(chimaera_transport INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)

option(GEM5_TRANSPORT_BUILD_GUEST "Build guest library (requires custom libm5)" ON)
option(GEM5_TRANSPORT_BUILD_EXAMPLES "Build interactive examples" ${PROJECT_IS_TOP_LEVEL})
set(GEM5_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../gem5" CACHE PATH "Custom gem5 source tree")

add_library(gem5_host src/host_transport.cpp)
add_library(chimaera::host ALIAS gem5_host)
target_link_libraries(gem5_host PUBLIC chimaera::transport)
target_include_directories(gem5_host PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)

find_package(Threads REQUIRED)
if(NOT TARGET chimaera::queue_manager)
    add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../queue-manager ${CMAKE_CURRENT_BINARY_DIR}/queue-manager)
endif()
add_library(gem5_channels src/channel_service.cpp)
add_library(chimaera::channels ALIAS gem5_channels)
target_link_libraries(gem5_channels PUBLIC chimaera::transport chimaera::queue_manager PRIVATE Threads::Threads)

add_library(gem5_host_controller src/gem5_host_controller.cpp src/gem5_timing_controller.cpp
    src/wall_clock_pacer.cpp)
add_library(chimaera::host_controller ALIAS gem5_host_controller)
target_link_libraries(gem5_host_controller PUBLIC chimaera::host PRIVATE Threads::Threads)

if(GEM5_TRANSPORT_BUILD_GUEST)
    find_path(GEM5_OP_INCLUDE gem5/m5ops.h HINTS "${GEM5_ROOT}/include" REQUIRED)
    find_path(GEM5_MMAP_INCLUDE m5_mmap.h HINTS "${GEM5_ROOT}/util/m5/src" REQUIRED)
    find_library(GEM5_M5_LIBRARY m5 HINTS "${GEM5_ROOT}/util/m5/build/x86/out" REQUIRED)
    add_library(gem5_guest src/guest_transport.cpp)
    add_library(chimaera::guest ALIAS gem5_guest)
    target_include_directories(gem5_guest PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include
        PRIVATE ${GEM5_OP_INCLUDE} ${GEM5_MMAP_INCLUDE})
    target_link_libraries(gem5_guest PUBLIC chimaera::transport PRIVATE ${GEM5_M5_LIBRARY})
    target_link_options(gem5_guest INTERFACE -no-pie)
    add_library(gem5_guest_controller src/gem5_guest_controller.cpp)
    add_library(chimaera::guest_controller ALIAS gem5_guest_controller)
    target_link_libraries(gem5_guest_controller PUBLIC chimaera::guest PRIVATE Threads::Threads)
endif()

if(GEM5_TRANSPORT_BUILD_EXAMPLES)
    add_executable(gem5_transport_host examples/host.cpp)
    target_link_libraries(gem5_transport_host PRIVATE chimaera::host)
    add_executable(gem5_controller_host examples/controller_host.cpp)
    target_link_libraries(gem5_controller_host PRIVATE chimaera::host_controller chimaera::channels)
    if(GEM5_TRANSPORT_BUILD_GUEST)
        add_executable(gem5_transport_guest examples/guest.cpp)
        target_link_libraries(gem5_transport_guest PRIVATE chimaera::guest)
        target_include_directories(gem5_transport_guest PRIVATE ${GEM5_MMAP_INCLUDE})
        add_executable(gem5_controller_guest examples/controller_guest.cpp)
        target_link_libraries(gem5_controller_guest PRIVATE chimaera::guest_controller chimaera::channels)
        target_include_directories(gem5_controller_guest PRIVATE ${GEM5_MMAP_INCLUDE} ${GEM5_OP_INCLUDE})
    endif()
endif()

option(GEM5_TRANSPORT_BUILD_TESTS "Build channel integration tests" ${PROJECT_IS_TOP_LEVEL})
if(GEM5_TRANSPORT_BUILD_TESTS)
    enable_testing()
    add_executable(channel_service_tests tests/channel_service_tests.cpp)
    target_link_libraries(channel_service_tests PRIVATE chimaera::channels Threads::Threads)
    add_executable(guest_channels_tests tests/guest_channels_tests.cpp)
    target_link_libraries(guest_channels_tests PRIVATE chimaera::channels)
    add_test(NAME guest_channels COMMAND guest_channels_tests)
    set_tests_properties(guest_channels PROPERTIES TIMEOUT 15)
    add_test(NAME channel_service COMMAND channel_service_tests)
    set_tests_properties(channel_service PROPERTIES TIMEOUT 15)
endif()

if(GEM5_TRANSPORT_BUILD_TESTS AND GEM5_TRANSPORT_BUILD_EXAMPLES)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
        set(channel_console_programs $<TARGET_FILE:gem5_controller_host>)
        add_test(NAME channel_console COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tests/channel_console_test.py ${channel_console_programs})
        set_tests_properties(channel_console PROPERTIES TIMEOUT 20)
    endif()
endif()
