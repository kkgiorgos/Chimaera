cmake_minimum_required(VERSION 3.21)
project(Gem5Transport LANGUAGES CXX)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The gem5 transport requires Linux")
endif()

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_library(chimaera_transport INTERFACE)
add_library(chimaera::transport ALIAS chimaera_transport)
target_compile_features(chimaera_transport INTERFACE cxx_std_20)
target_include_directories(chimaera_transport INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)

option(GEM5_TRANSPORT_BUILD_GUEST "Build guest library (requires custom libm5)" ON)
set(GEM5_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../gem5" CACHE PATH "Custom gem5 source tree")

add_library(gem5_host src/host_transport.cpp)
add_library(chimaera::host ALIAS gem5_host)
target_link_libraries(gem5_host PUBLIC chimaera::transport)

find_package(Threads REQUIRED)
add_library(gem5_host_controller src/gem5_host_controller.cpp src/gem5_timing_controller.cpp
    src/dma_host_controller.cpp src/m5_host_controller.cpp)
target_include_directories(gem5_host_controller PRIVATE "${GEM5_ROOT}/include")
add_library(chimaera::host_controller ALIAS gem5_host_controller)
target_link_libraries(gem5_host_controller PUBLIC chimaera::host PRIVATE Threads::Threads)

if(GEM5_TRANSPORT_BUILD_GUEST)
    find_path(GEM5_OP_INCLUDE gem5/m5ops.h HINTS "${GEM5_ROOT}/include" REQUIRED)
    find_path(GEM5_MMAP_INCLUDE m5_mmap.h HINTS "${GEM5_ROOT}/util/m5/src" REQUIRED)
    find_library(GEM5_M5_LIBRARY m5 HINTS "${GEM5_ROOT}/util/m5/build/x86/out" REQUIRED)
    add_library(gem5_guest src/guest_transport.cpp)
    add_library(chimaera::guest ALIAS gem5_guest)
    target_include_directories(gem5_guest PRIVATE ${GEM5_OP_INCLUDE} ${GEM5_MMAP_INCLUDE})
    target_link_libraries(gem5_guest PUBLIC chimaera::transport PRIVATE ${GEM5_M5_LIBRARY})
    target_link_options(gem5_guest INTERFACE -no-pie)
    add_library(gem5_guest_controller src/gem5_guest_controller.cpp src/dma_guest_controller.cpp)
    target_include_directories(gem5_guest_controller PRIVATE "${GEM5_ROOT}/include")
    add_library(chimaera::guest_controller ALIAS gem5_guest_controller)
    target_link_libraries(gem5_guest_controller PUBLIC chimaera::guest PRIVATE Threads::Threads)

    # The DDS plugin is a shared library. Compile PIC m5 wrappers directly for
    # this endpoint instead of inheriting the executable-only libm5 -no-pie ABI.
    enable_language(C ASM)
    add_library(chimaera_m5_endpoint STATIC src/m5_endpoint.cpp
        "${GEM5_ROOT}/util/m5/src/m5_mmap.c"
        "${GEM5_ROOT}/util/m5/src/abi/x86/m5op.S"
        "${GEM5_ROOT}/util/m5/src/abi/x86/m5op_addr.S")
    add_library(chimaera::m5_endpoint ALIAS chimaera_m5_endpoint)
    set_target_properties(chimaera_m5_endpoint PROPERTIES POSITION_INDEPENDENT_CODE ON)
    target_compile_definitions(chimaera_m5_endpoint PRIVATE M5OP_PIC)
    target_include_directories(chimaera_m5_endpoint PUBLIC ${GEM5_MMAP_INCLUDE}
        PRIVATE ${GEM5_OP_INCLUDE})
    target_link_libraries(chimaera_m5_endpoint PUBLIC chimaera::transport PRIVATE Threads::Threads)
endif()

option(GEM5_TRANSPORT_BUILD_TESTS "Build transport and controller tests" ${PROJECT_IS_TOP_LEVEL})
if(GEM5_TRANSPORT_BUILD_TESTS)
    enable_testing()
    add_executable(m5_checkpoint_tests tests/m5_checkpoint_test.cpp)
    target_include_directories(m5_checkpoint_tests PRIVATE ${GEM5_ROOT}/include)
    target_compile_features(m5_checkpoint_tests PRIVATE cxx_std_20)
    target_compile_options(m5_checkpoint_tests PRIVATE -UNDEBUG)
    add_test(NAME m5_checkpoint COMMAND m5_checkpoint_tests)
    add_executable(message_tests tests/message_tests.cpp)
    add_executable(dma_protocol_tests tests/dma_protocol_tests.cpp)
    target_link_libraries(dma_protocol_tests PRIVATE chimaera::transport)
    add_test(NAME dma_protocol COMMAND dma_protocol_tests)
    target_link_libraries(message_tests PRIVATE chimaera::transport)
    add_test(NAME message COMMAND message_tests)
    add_executable(mailbox_tests tests/mailbox_tests.cpp)
    target_include_directories(mailbox_tests PRIVATE ${GEM5_ROOT}/include)
    target_compile_features(mailbox_tests PRIVATE cxx_std_20)
    add_test(NAME mailbox COMMAND mailbox_tests)
    set_tests_properties(mailbox PROPERTIES TIMEOUT 30)
    add_executable(host_controller_tests tests/host_controller_tests.cpp)
    add_executable(dma_host_tests tests/dma_host_tests.cpp)
    target_include_directories(dma_host_tests PRIVATE ${GEM5_ROOT}/include)
    target_link_libraries(dma_host_tests PRIVATE chimaera::host_controller Threads::Threads)
    add_test(NAME dma_host COMMAND dma_host_tests)
    set_tests_properties(dma_host PROPERTIES TIMEOUT 15)
    add_executable(m5_host_tests tests/m5_host_tests.cpp)
    target_include_directories(m5_host_tests PRIVATE ${GEM5_ROOT}/include)
    target_link_libraries(m5_host_tests PRIVATE chimaera::host_controller Threads::Threads)
    add_test(NAME m5_host COMMAND m5_host_tests)
    set_tests_properties(m5_host PROPERTIES TIMEOUT 30)
    target_link_libraries(host_controller_tests PRIVATE chimaera::host_controller Threads::Threads)
    add_test(NAME host_controller COMMAND host_controller_tests)
    set_tests_properties(host_controller PROPERTIES TIMEOUT 15)
    add_executable(controller_protocol_tests tests/controller_protocol_tests.cpp)
    target_link_libraries(controller_protocol_tests PRIVATE chimaera::transport)
    add_test(NAME controller_protocol COMMAND controller_protocol_tests)
    set_tests_properties(controller_protocol PROPERTIES TIMEOUT 15)
    add_executable(stream_batch_tests tests/stream_batch_tests.cpp)
    target_link_libraries(stream_batch_tests PRIVATE chimaera::transport)
    add_test(NAME stream_batch COMMAND stream_batch_tests)
    set_tests_properties(stream_batch PROPERTIES TIMEOUT 15)
    if(GEM5_TRANSPORT_BUILD_GUEST)
        add_executable(dma_guest_tests tests/dma_guest_tests.cpp)
        target_include_directories(dma_guest_tests PRIVATE ${GEM5_ROOT}/include)
        target_link_libraries(dma_guest_tests PRIVATE chimaera::guest_controller Threads::Threads)
        target_link_options(dma_guest_tests PRIVATE -Wl,--wrap=open -Wl,--wrap=close
            -Wl,--wrap=read -Wl,--wrap=mmap -Wl,--wrap=ioctl)
        add_test(NAME dma_guest COMMAND dma_guest_tests)
        set_tests_properties(dma_guest PROPERTIES TIMEOUT 20)
        add_executable(guest_m5ops_tests tests/guest_m5ops_tests.cpp
            src/guest_transport.cpp src/gem5_guest_controller.cpp)
        target_include_directories(guest_m5ops_tests PRIVATE ${GEM5_OP_INCLUDE} ${GEM5_MMAP_INCLUDE})
        target_link_libraries(guest_m5ops_tests PRIVATE chimaera::transport Threads::Threads)
        target_link_options(guest_m5ops_tests PRIVATE -Wl,--wrap=nanosleep)
        add_test(NAME guest_m5ops COMMAND guest_m5ops_tests)
        set_tests_properties(guest_m5ops PROPERTIES TIMEOUT 15)
        # Supply executable-local m5 stubs; do not pull in the endpoint's PIC
        # assembly archive or trigger real m5 operations on the native host.
        add_executable(m5_endpoint_tests tests/m5_endpoint_tests.cpp src/m5_endpoint.cpp)
        target_include_directories(m5_endpoint_tests PRIVATE ${GEM5_OP_INCLUDE} ${GEM5_MMAP_INCLUDE})
        target_link_libraries(m5_endpoint_tests PRIVATE chimaera::transport Threads::Threads)
        target_link_options(m5_endpoint_tests PRIVATE -Wl,--wrap=open -Wl,--wrap=close
            -Wl,--wrap=read -Wl,--wrap=mmap -Wl,--wrap=munmap -Wl,--wrap=ioctl)
        add_test(NAME m5_endpoint COMMAND m5_endpoint_tests)
        set_tests_properties(m5_endpoint PROPERTIES TIMEOUT 15)
    endif()
endif()
